⚡ FLEXBOX — PARENT (Container) vs CHILD (Item)
PARENT flex container
.container {
  display: flex;
  /* Arah */
  flex-direction: row | row-reverse
                | column | column-reverse;
  /* Wrap */
  flex-wrap: nowrap | wrap | wrap-reverse;
  /* Shorthand: direction + wrap */
  flex-flow: row wrap;

  /* Penyelarasan Horizontal */
  justify-content: flex-start | flex-end
    | center | space-between
    | space-around | space-evenly;

  /* Penyelarasan Vertikal */
  align-items: stretch | flex-start
    | flex-end | center | baseline;

  /* Multi-line alignment */
  align-content: flex-start | center
    | space-between | stretch;

  gap: 10px;           /* jarak antar item */
  row-gap: 10px;
  column-gap: 20px;
}
CHILD flex item
.item {
  /* Urutan tampil */
  order: 0;       /* default 0 */

  /* Kemampuan membesar */
  flex-grow: 0;   /* default 0 */
  flex-grow: 1;   /* isi sisa ruang */

  /* Kemampuan mengecil */
  flex-shrink: 1; /* default 1 */

  /* Ukuran awal */
  flex-basis: auto | 200px | 50%;

  /* Shorthand: grow shrink basis */
  flex: 1;         /* = 1 1 0% */
  flex: 1 1 auto;
  flex: 0 0 200px; /* fixed width */

  /* Override align parent */
  align-self: auto | flex-start
    | flex-end | center | stretch;
}
Trik Umum
/* Center sempurna */
.parent { display:flex;
  justify-content:center;
  align-items:center; }

/* Equal columns */
.item { flex: 1; }

/* Push to right */
.last { margin-left: auto; }

@media (min-width: 768px) { /* tablet+ */ }

@media (max-width: 768px) { /* tablet- */ }

/* Orientasi */
@media (orientation: landscape) {}
@media (orientation: portrait) {}

🏆 PRIORITY CODE — CARD FLEX + LANDING PAGE SEMANTIC
CODE CARD — .card { display: flex }
/* ── HTML ── */
<div class="card">
  <img class="card__img" src="img.jpg" alt="foto">
  <div class="card__body">
    <h3 class="card__title">Judul Card</h3>
    <p class="card__text">Deskripsi singkat.</p>
    <a href="#" class="card__btn">Selengkapnya</a>
  </div>
</div>

/* ── CSS ── */
.card {
  display: flex;          /* horizontal layout */
  flex-direction: row;    /* default */
  gap: 16px;
  border: 1px solid #ddd;
  border-radius: 8px;
  overflow: hidden;
  max-width: 480px;
}
.card__img {
  width: 150px;
  object-fit: cover;      /* crop proporsional */
  flex-shrink: 0;         /* jangan mengecil */
}
.card__body {
  display: flex;
  flex-direction: column; /* vertikal */
  padding: 12px;
  gap: 6px;
}
.card__btn {
  margin-top: auto;       /* push ke bawah */
  padding: 6px 14px;
  background: #000;
  color: #fff;
  text-decoration: none;
  border-radius: 4px;
  align-self: flex-start;
}
/* Responsive: stack vertical di mobile */
@media (max-width: 480px) {
  .card { flex-direction: column; }
  .card__img { width: 100%; height: 160px; }
}

<!-- HTML SEMANTIC -->
<body>
  <header>
    <nav>
      <a href="#" class="logo">Brand</a>
      <ul class="nav__list">
        <li><a href="#about">About</a></li>
        <li><a href="#work">Work</a></li>
        <li><a href="#contact">Contact</a></li>
      </ul>
    </nav>
  </header>

  <main>
    <section id="hero">
      <h1>Headline Utama</h1>
      <p>Subheadline / tagline.</p>
      <a href="#" class="btn">CTA Button</a>
    </section>

    <section id="work">
      <article class="card">...</article>
      <article class="card">...</article>
    </section>
  </main>

  <footer><p>&copy; 2025 Brand</p></footer>
</body>

/* ── CSS MEDIA QUERY TERPADU ── */
* { box-sizing: border-box; margin: 0; padding: 0; }

/* Mobile First — Base Styles */
nav    { display: flex; flex-direction: column;
          align-items: center; padding: 12px; }
.nav__list { display: none; }          /* hidden mobile */
#hero  { text-align: center; padding: 40px 16px; }
#work  { display: flex; flex-direction: column;
          gap: 16px; padding: 24px 16px; }
.card  { border: 1px solid #ddd; padding: 16px; }

/* Tablet ≥ 768px */
@media (min-width: 768px) {
  nav        { flex-direction: row;
               justify-content: space-between; }
  .nav__list { display: flex; gap: 20px;
               list-style: none; }
  #work      { flex-direction: row; flex-wrap: wrap; }
  .card      { flex: 1 1 calc(50% - 16px); }
}

/* Desktop ≥ 1024px */
@media (min-width: 1024px) {
  #hero  { padding: 80px 40px; }
  .card  { flex: 1 1 calc(33.33% - 16px); }
  #work  { max-width: 1200px; margin: 0 auto; }
}

🎨 SYNTAX CSS — ANATOMI RULE & PROPERTI PENTING
Anatomi CSS Rule
/* selector { properti: nilai; } */

h1 {                    /* selector */
  color: #333;           /* deklarasi */
  font-size: 2rem;       /* properti: nilai */
  margin: 0 auto;        /* shorthand */
}

/* Multiple selector */
h1, h2, h3 { color: #000; }

/* Combinator */
div p     { }   /* descendant */
ul > li   { }   /* child langsung */
h1 + p    { }   /* saudara setelah */
h1 ~ p    { }   /* semua saudara setelah */
Warna & Background
.elemen {
  color: #ff0000;          /* hex */
  color: rgb(255, 0, 0);   /* rgb */
  color: rgba(0,0,0,0.5);  /* + opacity */
  color: hsl(0, 100%, 50%); /* hsl */

  background-color: #fff;
  background-image: url('bg.jpg');
  background-size: cover | contain | auto;
  background-position: center center;
  background-repeat: no-repeat | repeat;

  /* Shorthand */
  background: #000 url('bg.jpg')
              no-repeat center/cover;
  opacity: 0.8;            /* 0=bening,1=solid */
}
Positioning
.box {
  position: static;    /* default */

  top: 0; right: 0;
  bottom: 0; left: 0;

}
CSS Variable (Custom Properties)
